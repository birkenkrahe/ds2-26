  #+begin_src R :file ../../img/palmer3.png :session *R* :results graphics output file :exports both
    if (!is.null(df)) {

      species <- unique(df$species)
      cols <- c("blue", "red", "darkgreen")

      plot(df$bill_length_mm,
           df$bill_depth_mm,
           type = "n",
           xlab = "Bill length (mm)",
           ylab = "Bill depth (mm)",
           main = "Palmer Penguins by Species")

      for (i in seq_along(species)) {
        sub <- df[df$species == species[i], ]
        points(sub$bill_length_mm,
               sub$bill_depth_mm,
               col = cols[i],
               pch = 19)
      }

      legend("bottomright",
             legend = species,
             col = cols,
             pch = 19)
    }
  #+end_src
